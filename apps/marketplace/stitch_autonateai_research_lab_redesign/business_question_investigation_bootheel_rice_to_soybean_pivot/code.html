Acreage Enterprise Budget Delta Query

Calculates per-acre net margin spreads by joining USDA RMA premium structures, alluvial lift electrical tariffs, and wholesale localized chemical indexes: 

WITH regional_budgets AS (
  SELECT 
    p.county_fips,
    p.soil_hydrology_rating,
    c.crop_type,
    (c.expected_yield_bu * c.contract_price) AS gross_revenue,
    (c.seed_cost + c.n_p_k_cost + c.pumping_tariff_cost + c.drying_cost) AS total_variable_cost,
    c.ins_liability_limit
  FROM telemetry.missouri_bootheel_parcels p
  JOIN models.enterprise_budgets_2026 c ON p.parcel_id = c.parcel_id
  WHERE p.county_name IN ('Dunklin', 'Pemiscot', 'Stoddard')
)
SELECT 
  county_fips,
  ROUND(AVG(CASE WHEN crop_type = 'SOY_GRP_IV' THEN (gross_revenue - total_variable_cost) END), 2) AS soybean_net_margin,
  ROUND(AVG(CASE WHEN crop_type = 'RICE_CONV' THEN (gross_revenue - total_variable_cost) END), 2) AS rice_net_margin,
  ROUND(AVG(ins_liability_limit), 2) AS aggregate_liability_shift
FROM regional_budgets
GROUP BY county_fips
ORDER BY county_fips ASC;
